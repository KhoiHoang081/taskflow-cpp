cmake_minimum_required(VERSION 3.16)
project(TaskFlow VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(taskflow_core src/TaskManager.cpp src/FileStorage.cpp)
target_include_directories(taskflow_core PUBLIC include)
if(MSVC)
  target_compile_options(taskflow_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(taskflow_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(taskflow src/main.cpp src/Console.cpp)
target_link_libraries(taskflow PRIVATE taskflow_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(taskflow_tests tests/test_main.cpp)
  target_link_libraries(taskflow_tests PRIVATE taskflow_core)
  add_test(NAME taskflow_tests COMMAND taskflow_tests)
endif()
